# 操作日志审计功能实现方案

## 1.1 设计目标

通过注解驱动 + AOP 拦截的方式，实现精准的操作日志记录，满足以下需求：
- 订单、工单等核心业务操作留痕
- 满足监管合规要求
- 便于问题追溯和审计

## 1.2 技术方案

**核心设计**：采用「注解驱动 + AOP 切面 + 异步写入」方案。

```
┌──────────────────────────────────────────────────────────────┐
│                        操作日志流程                            │
├──────────────────────────────────────────────────────────────┤
│                                                              │
│  @OperationLog(module="订单管理", operation="CREATE",          │
│                description="创建医疗器械订单")                  │
│                           │                                 │
│                           ▼                                 │
│  ┌─────────────────────────────────────────────┐            │
│  │              OperationLogAspect               │            │
│  │  1. Around 拦截带注解的方法                   │            │
│  │  2. 获取当前登录用户                          │            │
│  │  3. 解析 @OperationLog 参数                   │            │
│  │  4. 获取请求参数（脱敏处理）                  │            │
│  │  5. 异步写入 sys_operation_log               │            │
│  └─────────────────────────────────────────────┘            │
│                           │                                 │
│                           ▼                                 │
│  sys_operation_log ── 日志查询 / Excel导出                    │
│                                                              │
└──────────────────────────────────────────────────────────────┘
```

## 1.3 表结构

```sql
CREATE TABLE sys_operation_log (
    id              BIGINT          NOT NULL AUTO_INCREMENT COMMENT '主键ID',
    module         VARCHAR(64)     NOT NULL COMMENT '模块（如：订单管理、工单管理）',
    business_type  VARCHAR(64)     NOT NULL COMMENT '业务类型编码（如：erp_order）',
    business_id    VARCHAR(64)    COMMENT '业务ID（主键值，如订单ID）',
    operation      VARCHAR(64)     NOT NULL COMMENT '操作类型（CREATE/UPDATE/DELETE/AUDIT）',
    operation_desc VARCHAR(256)    COMMENT '操作描述（如：创建医疗器械订单）',
    request_method VARCHAR(10)     COMMENT 'HTTP请求方法（GET/POST/PUT/DELETE）',
    request_url    VARCHAR(512)    COMMENT '请求URL',
    request_params TEXT            COMMENT '请求参数（JSON格式，已脱敏）',
    request_ip     VARCHAR(128)   COMMENT '请求IP地址',
    user_id        BIGINT          COMMENT '操作用户ID',
    user_name      VARCHAR(64)    COMMENT '操作用户姓名',
    operate_time   DATETIME        DEFAULT CURRENT_TIMESTAMP COMMENT '操作时间',
    cost_time      BIGINT          COMMENT '请求耗时（毫秒）',
    PRIMARY KEY (id),
    KEY idx_log_module (module),
    KEY idx_log_business_type (business_type),
    KEY idx_log_business_id (business_id),
    KEY idx_log_user_id (user_id),
    KEY idx_log_operate_time (operate_time),
    KEY idx_log_operation (operation)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='操作日志表';
```

## 1.4 代码设计

### 1.4.1 注解定义

```java
// 路径: yigongbao-framework/src/main/java/com/yigongbao/framework/annotation/OperationLog.java

package com.yigongbao.framework.annotation;

import java.lang.annotation.*;

/**
 * 操作日志注解
 * 标注在 Controller 方法上，自动记录操作日志
 *
 * @author hanjor
 * @date 2026-03-24
 */
@Target(ElementType.METHOD)
@Retention(RetentionPolicy.RUNTIME)
@Documented
public @interface OperationLog {

    /**
     * 模块名称
     * 如：订单管理、工单管理、生产管理
     */
    String module();

    /**
     * 业务类型编码
     * 如：erp_order、erp_work_order
     */
    String businessType();

    /**
     * 操作类型
     * 枚举值：CREATE / UPDATE / DELETE / AUDIT / EXPORT / IMPORT
     */
    String operation();

    /**
     * 操作描述
     * 支持 SpEL 表达式，如：创建订单，订单号={#orderNo}
     */
    String description();

    /**
     * 是否记录请求参数
     * 默认 true
     */
    boolean logParams() default true;

    /**
     * 是否记录返回结果
     * 默认 false（大型数据不建议开启）
     */
    boolean logResult() default false;
}
```

### 1.4.3 AOP 切面

```java
// 路径: yigongbao-framework/src/main/java/com/yigongbao/framework/aspect/OperationLogAspect.java

package com.yigongbao.framework.aspect;

import cn.dev33.satoken.stp.StpUtil;
import com.alibaba.fastjson2.JSON;
import com.yigongbao.common.exception.BusinessException;
import com.yigongbao.framework.annotation.OperationLog;
import com.yigongbao.module.system.operationlog.entity.OperationLogEntity;
import com.yigongbao.module.system.operationlog.service.OperationLogService;
import jakarta.servlet.http.HttpServletRequest;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.aspectj.lang.ProceedingJoinPoint;
import org.aspectj.lang.annotation.Around;
import org.aspectj.lang.annotation.Aspect;
import org.aspectj.lang.reflect.MethodSignature;
import org.springframework.scheduling.annotation.Async;
import org.springframework.stereotype.Component;
import org.springframework.web.context.request.RequestContextHolder;
import org.springframework.web.context.request.ServletRequestAttributes;

import java.lang.reflect.Method;
import java.time.LocalDateTime;
import java.util.Map;
import java.util.regex.Pattern;

/**
 * 操作日志切面
 * 拦截标注 @OperationLog 的方法，记录操作日志
 *
 * @author hanjor
 * @date 2026-03-24
 */
@Component
@Aspect
@RequiredArgsConstructor
@Slf4j
public class OperationLogAspect {

    private final OperationLogService operationLogService;

    // 敏感字段脱敏正则
    private static final Pattern SENSITIVE_PATTERN = Pattern.compile(
        "\"?(password|passwd|secret|token|key|证书密码)\"?:\\s*\"[^\"]+\"",
        Pattern.CASE_INSENSITIVE);

    @Around("@annotation(com.yigongbao.framework.annotation.OperationLog)")
    public Object around(ProceedingJoinPoint point) throws Throwable {
        long startTime = System.currentTimeMillis();
        MethodSignature signature = (MethodSignature) point.getSignature();
        Method method = signature.getMethod();
        OperationLog annotation = method.getAnnotation(OperationLog.class);

        // 获取请求信息
        HttpServletRequest request = getRequest();
        String requestUrl = request != null ? request.getRequestURI() : "";
        String requestMethod = request != null ? request.getMethod() : "";
        String requestIp = getClientIp(request);

        // 获取当前用户
        Long userId = null;
        String userName = "系统";
        try {
            if (StpUtil.isLogin()) {
                userId = StpUtil.getLoginIdAsLong();
                userName = getUserName(userId);
            }
        } catch (Exception e) {
            log.debug("获取登录用户信息失败", e);
        }

        // 获取请求参数（脱敏）
        String requestParams = null;
        if (annotation.logParams()) {
            requestParams = getRequestParams(point, annotation);
        }

        // 执行目标方法
        boolean success = true;
        try {
            return point.proceed();
        } catch (BusinessException e) {
            success = false;
            throw e;
        } finally {
            long costTime = System.currentTimeMillis() - startTime;
            if (success) {
                saveLogAsync(annotation, requestUrl, requestMethod, requestIp,
                    userId, userName, requestParams, costTime);
            }
        }
    }

    @Async
    public void saveLogAsync(OperationLog annotation, String requestUrl,
            String requestMethod, String requestIp, Long userId, String userName,
            String requestParams, long costTime) {
        try {
            OperationLogEntity logEntity = new OperationLogEntity();
            logEntity.setModule(annotation.module());
            logEntity.setBusinessType(annotation.businessType());
            logEntity.setOperation(annotation.operation());
            logEntity.setOperationDesc(annotation.description());
            logEntity.setRequestMethod(requestMethod);
            logEntity.setRequestUrl(requestUrl);
            logEntity.setRequestParams(requestParams);
            logEntity.setRequestIp(requestIp);
            logEntity.setUserId(userId);
            logEntity.setUserName(userName);
            logEntity.setOperateTime(LocalDateTime.now());
            logEntity.setCostTime(costTime);

            operationLogService.save(logEntity);
        } catch (Exception e) {
            log.error("保存操作日志失败", e);
        }
    }

    private String maskSensitiveData(String json) {
        if (json == null) return null;
        return SENSITIVE_PATTERN.matcher(json).replaceAll("\"$1\":\"***\"");
    }

    private String getRequestParams(ProceedingJoinPoint point, OperationLog annotation) {
        try {
            Object[] args = point.getArgs();
            if (args == null || args.length == 0) return null;

            MethodSignature signature = (MethodSignature) point.getSignature();
            String[] paramNames = signature.getParameterNames();
            Map<String, Object> paramMap = new java.util.HashMap<>();
            for (int i = 0; i < args.length; i++) {
                if (args[i] instanceof HttpServletRequest
                    || args[i] instanceof jakarta.servlet.http.HttpServletResponse) {
                    continue;
                }
                String name = paramNames != null && i < paramNames.length
                    ? paramNames[i] : "arg" + i;
                paramMap.put(name, args[i]);
            }
            String json = JSON.toJSONString(paramMap);
            return maskSensitiveData(json);
        } catch (Exception e) {
            return null;
        }
    }

    private String getClientIp(HttpServletRequest request) {
        if (request == null) return null;
        String ip = request.getHeader("X-Forwarded-For");
        if (ip == null || ip.isEmpty()) {
            ip = request.getHeader("X-Real-IP");
        }
        if (ip == null || ip.isEmpty()) {
            ip = request.getRemoteAddr();
        }
        return ip;
    }

    private HttpServletRequest getRequest() {
        ServletRequestAttributes attrs = (ServletRequestAttributes) RequestContextHolder.getRequestAttributes();
        return attrs != null ? attrs.getRequest() : null;
    }

    private String getUserName(Long userId) {
        return "用户-" + userId;
    }
}
```

## 1.5 使用示例

```java
/**
 * 订单管理 Controller
 */
@RestController
@RequestMapping("/api/erp/order")
@RequiredArgsConstructor
public class OrderController {

    private final OrderService orderService;

    /**
     * 创建订单
     */
    @OperationLog(
        module = "订单管理",
        businessType = "erp_order",
        operation = "CREATE",
        description = "创建医疗器械订单"
    )
    @PostMapping
    public Result<OrderVO> create(@Validated @RequestBody CreateOrderDTO dto) {
        OrderVO order = orderService.create(dto);
        return Result.success(order);
    }

    /**
     * 审核订单
     */
    @OperationLog(
        module = "订单管理",
        businessType = "erp_order",
        operation = "AUDIT",
        description = "审核订单"
    )
    @PutMapping("/{id}/audit")
    public Result<Void> audit(@PathVariable Long id, @RequestBody AuditOrderDTO dto) {
        orderService.audit(id, dto);
        return Result.success();
    }
}
```
