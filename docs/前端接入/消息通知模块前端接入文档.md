```typescript
import $axios from '@/utils/http'
import type { NotificationMessage, MessageQueryParams, UnreadCount } from '@/types/notification'

/** 分页响应 */
interface PageResult<T> {
  records: T[]
  total: number
  size: number
  current: number
  pages: number
}

/** 查询消息列表 */
export function getNotificationList(params: MessageQueryParams) {
  return $axios.post<PageResult<NotificationMessage>>('/notification/messages', params)
}

/** 获取未读数量 */
export function getUnreadCount() {
  return $axios.get<UnreadCount>('/notification/messages/unread-count')
}

/** 标记单条已读 */
export function markMessageRead(id: number) {
  return $axios.put(`/notification/messages/${id}/read`)
}

/** 批量标记已读 */
export function batchMarkRead(data: { ids?: number[]; category?: string; markAll?: boolean }) {
  return $axios.put('/notification/messages/batch-read', data)
}

/** 确认弹窗消息 */
export function confirmMessage(id: number) {
  return $axios.put(`/notification/messages/${id}/confirm`)
}

/** 删除消息 */
export function deleteMessage(id: number) {
  return $axios.delete(`/notification/messages/${id}`)
```

### 2.3 WebSocket 工具类

**src/utils/websocket.ts**

```typescript
import { Session } from '@/utils/storage'

export class NotificationWebSocket {
  private ws: WebSocket | null = null
  private reconnectTimer: number | null = null
  private heartbeatTimer: number | null = null
  private isManualClose = false

  constructor(
    private url: string,
    private onMessage: (data: any) => void
  ) {}

  connect() {
    if (this.ws?.readyState === WebSocket.OPEN) return

    const token = Session.get('token')
    if (!token) return

    this.isManualClose = false
    this.ws = new WebSocket(this.url)

    this.ws.onopen = () => {
      console.log('[WS] 连接成功')
      this.send({ type: 'AUTH', token })
      this.startHeartbeat()
    }

    this.ws.onmessage = (e) => {
      try {
        const data = JSON.parse(e.data)
        this.onMessage(data)
      } catch (err) {
        console.error('[WS] 消息解析失败', err)
      }
    }

    this.ws.onerror = (err) => {
      console.error('[WS] 连接错误', err)
    }

    this.ws.onclose = () => {
      console.log('[WS] 连接关闭')
      this.stopHeartbeat()
      if (!this.isManualClose) {
        this.reconnectTimer = window.setTimeout(() => this.connect(), 3000)
      }
    }
  }

  send(data: any) {
    if (this.ws?.readyState === WebSocket.OPEN) {
      this.ws.send(JSON.stringify(data))
    }
  }

  private startHeartbeat() {
    this.heartbeatTimer = window.setInterval(() => {
      this.send({ type: 'PING' })
    }, 30000)
  }

  private stopHeartbeat() {
    if (this.heartbeatTimer) {
      clearInterval(this.heartbeatTimer)
      this.heartbeatTimer = null
    }
  }

  disconnect() {
    this.isManualClose = true
    if (this.reconnectTimer) {
      clearTimeout(this.reconnectTimer)
      this.reconnectTimer = null
    }
    this.stopHeartbeat()
    this.ws?.close()
    this.ws = null
  }
}
```

### 2.4 Pinia 状态管理

**src/stores/notification.ts**

```typescript
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import { getNotificationList, getUnreadCount, confirmMessage } from '@/api/notification'
import { NotificationWebSocket } from '@/utils/websocket'
import type { NotificationMessage, WebSocketMessage } from '@/types/notification'

export const useNotificationStore = defineStore('notification', () => {
  const unreadCount = ref(0)
  const popupQueue = ref<NotificationMessage[]>([])
  const currentPopup = ref<NotificationMessage | null>(null)
  
  let wsInstance: NotificationWebSocket | null = null

  const hasUnread = computed(() => unreadCount.value > 0)

  /** 初始化通知系统 */
  async function init() {
    // 1. 刷新未读数
    await refreshUnreadCount()

    // 2. 拉取离线弹窗消息
    try {
      const res = await getNotificationList({
        messageType: 'POPUP',
        isConfirmed: 0,
        pageSize: 100
      })
      if (res.records?.length) {
        popupQueue.value = res.records
        showNextPopup()
      }
    } catch (err) {
      console.error('拉取离线弹窗失败', err)
    }

    // 3. 启动 WebSocket
    connectWebSocket()
  }

  /** 刷新未读数 */
  async function refreshUnreadCount() {
    try {
      const res = await getUnreadCount()
      unreadCount.value = res.total
    } catch (err) {
      console.error('获取未读数失败', err)
    }
  }

  /** 连接 WebSocket */
  function connectWebSocket() {
    const wsUrl = import.meta.env.VITE_WS_URL || 'ws://localhost:8080/yi/notification/websocket'
    
    wsInstance = new NotificationWebSocket(wsUrl, handleWebSocketMessage)
    wsInstance.connect()
  }

  /** WebSocket 消息处理 */
  function handleWebSocketMessage(msg: WebSocketMessage) {
    if (msg.type === 'NEW_MESSAGE') {
      refreshUnreadCount()

      if (msg.data.messageType === 'POPUP') {
        popupQueue.value.push(msg.data)
        if (!currentPopup.value) {
          showNextPopup()
        }
      }
    }
  }

  /** 显示下一个弹窗 */
  function showNextPopup() {
    if (popupQueue.value.length === 0) {
      currentPopup.value = null
      return
    }
    currentPopup.value = popupQueue.value.shift()!
  }

  /** 确认当前弹窗 */
  async function confirmCurrentPopup() {
    if (!currentPopup.value) return

    try {
      await confirmMessage(currentPopup.value.id)
      await refreshUnreadCount()
    } catch (err) {
      console.error('确认弹窗失败', err)
    } finally {
      showNextPopup()
    }
  }

  /** 断开 WebSocket */
  function disconnect() {
    wsInstance?.disconnect()
    wsInstance = null
  }

  return {
    unreadCount,
    hasUnread,
    currentPopup,
    init,
    refreshUnreadCount,
    confirmCurrentPopup,
    disconnect
  }
})
```

```vue
<template>
  <div class="notification-list">
    <el-tabs v-model="activeTab" @tab-change="handleTabChange">
      <el-tab-pane label="全部" name="" />
      <el-tab-pane label="订单" name="ORDER" />
      <el-tab-pane label="审核" name="APPROVAL" />
      <el-tab-pane label="生产" name="PRODUCTION" />
    </el-tabs>

    <el-scrollbar height="calc(100vh - 200px)">
      <div
        v-for="msg in messages"
        :key="msg.id"
        :class="['msg-item', { unread: msg.isRead === 0 }]"
        @click="handleClick(msg)"
      >
        <div class="msg-header">
          <span class="msg-title">{{ msg.title }}</span>
          <span class="msg-time">{{ formatTime(msg.createTime) }}</span>
        </div>
        <div v-if="msg.content" class="msg-content">{{ msg.content }}</div>
      </div>

      <el-empty v-if="!messages.length" description="暂无消息" />
    </el-scrollbar>

    <el-pagination
      v-if="total > 0"
      v-model:current-page="pageNum"
      v-model:page-size="pageSize"
      :total="total"
      layout="total, prev, pager, next"
      small
      @current-change="loadMessages"
    />
  </div>
</template>

<script setup lang="ts">
import { ref, onMounted } from 'vue'
import { useRouter } from 'vue-router'
import dayjs from 'dayjs'
import { getNotificationList, markMessageRead } from '@/api/notification'
import { useNotificationStore } from '@/stores/notification'
import type { NotificationMessage } from '@/types/notification'

const router = useRouter()
const notificationStore = useNotificationStore()

const activeTab = ref('')
const messages = ref<NotificationMessage[]>([])
const pageNum = ref(1)
const pageSize = ref(20)
const total = ref(0)

async function loadMessages() {
  try {
    const res = await getNotificationList({
      category: activeTab.value || undefined,
      pageNum: pageNum.value,
      pageSize: pageSize.value
    })
    messages.value = res.records || []
    total.value = res.total || 0
  } catch (err) {
    console.error('加载消息失败', err)
  }
}

function handleTabChange() {
  pageNum.value = 1
  loadMessages()
}

async function handleClick(msg: NotificationMessage) {
  // 标记已读
  if (msg.isRead === 0) {
    try {
      await markMessageRead(msg.id)
      msg.isRead = 1
      notificationStore.refreshUnreadCount()
    } catch (err) {
      console.error('标记已读失败', err)
    }
  }

  // 跳转业务页面
  if (msg.jumpUrl) {
    router.push(msg.jumpUrl)
  }
}

function formatTime(time: string) {
  return dayjs(time).format('MM-DD HH:mm')
}

onMounted(loadMessages)
</script>
```

```vue
<template>
  <el-dialog
    v-model="visible"
    :title="currentPopup?.title"

    :show-close="false"
  >

      <el-button @click="handleViewDetail">查看详情</el-button>
      <el-button type="primary" @click="handleConfirm">确认</el-button>

<script setup lang="ts">
import { computed } from 'vue'
import { useRouter } from 'vue-router'

}

  notificationStore.confirmCurrentPopup()
}
```

## 五、注意事项

### 5.1 Token 认证

WebSocket 连接建立后，通过 AUTH 消息发送 token（从 Session 中获取）

### 5.2 baseURL 配置

项目使用 `/yi` 作为 baseURL，所有接口都会自动添加此前缀

### 5.3 错误处理

错误统一由 `src/utils/http.ts` 拦截器处理，会自动弹出 ElMessage 提示

### 5.4 登录状态检查

仅在用户已登录（`userInfos.isLogin === true`）时初始化通知系统

### 5.5 组件卸载

在 `onUnmounted` 中断开 WebSocket 连接，避免内存泄漏
